python3 models/DepthAnythingV2/make_da2_dataset.py \
  --in envtest/ros/train_set --out training/datasets/data_da2 --compare

then set  dataset = data_da2  in training/config/train.txt and:

python3 training/train.py --config training/config/train.txt
